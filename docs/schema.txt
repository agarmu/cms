table photos_cms_internal.database_identity:
  singleton: boolean primary key
  id: uuid

  check database_identity_singleton: singleton

table cameras:
  id: uuid primary key
  name: text

table lenses:
  id: uuid primary key
  name: text

table photos:
  id: uuid primary key
  taken_at: timestamp with time zone
  posted_at: timestamp with time zone
  alt_text: text
  caption: text
  description: text
  location: text?
  visible: boolean
  camera_id: uuid? references cameras.id
  lens_id: uuid? references lenses.id

table image_assets:
  id: integer primary key generated always as identity
  photo_id: uuid references photos.id
  width: integer
  height: integer
  byte_size: bigint?
  asset_location: text

  check image_assets_width_positive: width > 0
  check image_assets_height_positive: height > 0
  check image_assets_byte_size_valid:
    byte_size > 0 AND byte_size <= 9007199254740991

table image_uploads:
  id: uuid primary key
  photo_id: uuid
  kind: text
  owner_issuer: text?
  owner_subject: text?
  created: timestamp with time zone
  assets: jsonb?
  consumed: timestamp with time zone?
  input_hash: text?
  expires: timestamp with time zone
  cleaned: timestamp with time zone?
  completed: timestamp with time zone?
  cancelled: timestamp with time zone?
  failed: text?
  retry_count: integer
  next_check: timestamp with time zone

  index image_uploads_photo_id_idx: (photo_id)
  index image_uploads_expiry_idx: (expires) WHERE cleaned IS NULL
  index image_uploads_pending_idx: (next_check)
    WHERE completed IS NULL AND failed IS NULL AND cancelled IS NULL

table image_asset_garbage:
  photo_id: uuid
  asset_location: text
  retry_count: integer
  failed: text?
  next_check: timestamp with time zone

  primary key: (photo_id, asset_location)
  index image_asset_garbage_next_check_idx: (next_check)
